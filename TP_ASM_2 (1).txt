MAIN PROC


;CRIAMOS UMA FUNCAO CHAMADA (INICIO) NA QUAL SERVIRA PARA
;CONSEGUIR FAZERMOS O LOOP CASO O PROGRAMA CHEGUE AO FIM e DESEJE CONTINUAR
INICIO:
    
    ;INICIALIZAR O SEGMENTO DATA
    MOV AX, @DATA
    MOV DS, AX

    ;CHAMAR A VARIAVEL MENU1
    MOV DX, OFFSET MENU1
    MOV AH, 9H
    INT 21H
    
    ;CHAMAR A VARIAVEL MENU2
    MOV DX, OFFSET MENU2
    MOV AH, 9H
    INT 21H

    ;CHAMAR A VARIAVEL MENU3
    MOV DX, OFFSET MENU3
    MOV AH, 9H
    INT 21H

    ;CHAMAR A VARIAVEL MENU4
    MOV DX, OFFSET MENU4
    MOV AH, 9H
    INT 21H

    ;CHAMAR A VARIAVEL MENU5    
    MOV DX, OFFSET MENU5
    MOV AH, 9H
    INT 21H

    ;CHAMAR A VARIAVEL MENUE NA QUAL `E O NOSSO MENU EXTRA    
    MOV DX, OFFSET MENUE
    MOV AH, 9H
    INT 21H


    ;CHAMAR A VARIAVEL MENU6
    MOV DX, OFFSET MENU6
    MOV AH, 9H
    INT 21H

    ;receber a opcao (1 OU 2 OU 3) E GUARDAR NO REGISTRO AL
    MOV AH, 1H
    INT 21H

    ;CHAMAR A VARIAVEL MENU7
    MOV DX, OFFSET MENU7
    MOV AH, 9H
    INT 21H
    
    ;CHAMAR A VARIAVEL MENU8
    MOV DX, OFFSET MENU8
    MOV AH, 9H
    INT 21H  

    ;PREFIRIMOS CRIAR UM METODO RESPONSAVEL SO PELA COMPARACAO (PARA SABER SE O VALOR DO AL SERA 1, 2 OU 3)
    JMP COMPARAR2 





;NOSSO METODO QUE SERA RESPONSAVEL POR COMPARAR OQUE O UTILIZADOR DIGITOU
COMPARAR2:

    ;COMPARAR OQUE ESTA NO AL COM 2
    CMP AL, '2'          ;PORQUE COMPARAR O VALOR DO AL COM 2? SIMPLES...
    ;SEM ESQUECER QUE DEVE SER COM ASPAS SIMPLES POIS O VALOR NO AL NAO `E EXATAMENTE 2, `E UM VALOR EM  HEXADECIMAL
    ;PRIMEIRO OBVIAMENTE SE FOR 2, LHE MANDAREMOS PARA A OPCAO 2
    JE OPCAO2

    ;SE O VALOR DO AL FOR MENOR QUE 2, TEREMOS QUE COMPARAR PARA SABER SE O VALOR INTRODUZIDO OU `E 1, OU `E NEGATIVO OU `E MENSAGEM VAZIA
    ;E SE FOR NEGATIVO, QUEREMOS MOSTRAR UMA MENSAGEM DE ERRO
    JB COMPARAR1

    ;SE O VALOR DO AL FOR MAIOR QUE 2, TEREMOS QUE COMPARAR PARA SABER SE O VALOR INTRODUZIDO OU `E 3 OU `E MAIOR
    ;E SE FOR MAIOR, QUEREMOS TAMBEM MOSTRAR UMA MENSAGEM DE ERRO
    JA COMPARAR3    


COMPARAR1:

    ;CONFIRMAR SE O VALOR `E 1, `E NEGATIVO
    CMP AL, '1' ;DEVE SER COM ASPAS SIMPLES POIS O VALOR NO AL NAO `E EXATAMENTE 1, `E HEXADECIMAL
    
    ;SE FOR 1, SALTA PARA A OPCAO 1
    JE OPCAO1

    ;SE NAO FOR 1, CONFIRMAR SE O UTILIZADOR SO DIGITOU ENTER (O ENTER TEM O VALOR 13 NO AL)
    ;O ENTER (13) `E MENOR QUE '2', POR ISSO E AQUI QUE ELE CHEGA E NAO NO COMPARAR3
    CMP AL, 13

    ;SE FOR MENSAGEM VAZIA (SO DIGITOU ENTER) MOSTRAR VAZIO
    JE VAZIO

    ;SE NAO FOR 1 NEM ENTER, ENTAO `E NEGATIVO, MOSTRAR ERRO
    JNE ERRO



COMPARAR3:

    ;CONFIRMAR SE O VALOR `E 3 OU `E MAIOR QUE 3
    CMP AL, '3' ;DEVE SER COM ASPAS SIMPLES POIS O VALOR NO AL NAO `E EXATAMENTE 3, `E HEXADECIMAL
    
    ;SE FOR 3, SALTA PARA A OPCAO 3
    JE OPCAO3

    ;SE FOR MAIOR, MOSTRAR ERRO
    JA ERRO

    ;A MENSAGEM VAZIA (SO ENTER) NUNCA CHEGA AQUI, POIS O ENTER (13) `E MENOR QUE '2' E JA FOI TRATADO NO COMPARAR1


;CRIAMOS UM METODO CHAMADO ERRO POIS CASO ALGO DE ERRO, SO CHAMAREMOS ESTA FUNCAO, AO INVEZ
;DE ESTAR SEMPRE A VISUALIZAR ERRO EM CADA PARTE DO NOSSO CODIGO
ERRO:

    ;MOSTRAR A MENSAGEM DO ERRO, E VOLTAR A LHE PEDIR A OPCAO CORRECTA
    MOV DX, OFFSET ERRO0
    MOV AH, 9H
    INT 21H

    ;RECEBER ESSA OPCAO
    MOV AH, 1H
    INT 21H

    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H

    ;LHE ENVIAR PARA O METODO DE COMPARACAO PARA VER SE DESTA VEZ DIGITOU A OPCAO CORRRECTA OU NAO
    JMP COMPARAR2


VAZIO:

    ;MOSTRAR A MENSAGEM PARA O UTILIZADOR VER QUE SABEMOS QUE A MENSAGEM DELE ESTA VAZIA, PARA ELE VOLTAR A INTRODUZIR UMA OPCAO CORRECTA
    MOV DX, OFFSET VAZIO1
    MOV AH, 9H
    INT 21H

    ;RECEBER ESSA OPCAO
    MOV AH, 1H
    INT 21H

    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H

    JMP COMPARAR2

OPCAO2:

    ;MOSTRAR AQUELA SETA DO MENU
    MOV DX, OFFSET MENU6
    MOV AH, 9H
    INT 21H


;AQUI FICARA A LOGICA DA CONVERSAO

    ;MOSTRAR A MENSAGEM QUE PEDE A FRASE AO UTILIZADOR (O 0AH ACEITA NO MAXIMO 80 CARACTERES)
    MOV DX, OFFSET PEDIR
    MOV AH, 9H
    INT 21H

    ;RECEBER A STRING
    ;O 0AH PRECISA QUE O DX APONTE PARA O INICIO DA VARIAVEL "RECEBER", SE NAO ELE GUARDA A STRING NO SITIO ERRADO
    MOV DX, OFFSET RECEBER
    MOV AH, 0AH
    INT 21H

    ;SALTAR PARA A LINHA DE BAIXO, PARA A ENTRADA E A SAIDA FICAREM EM LINHAS DIFERENTES
    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H

    ;O 0AH GUARDA NA POSICAO 1 DA VARIAVEL "RECEBER" A QUANTIDADE DE CARACTERES QUE O UTILIZADOR DIGITOU
    ;(A POSICAO 0 TEM O MAXIMO E A STRING COMECA NA POSICAO 2)
    ;COMPARAMOS ESSE VALOR COM 0 PARA SABER SE A FRASE ESTA VAZIA (SO DIGITOU ENTER)
    MOV AL, RECEBER+1
    CMP AL, 0

    ;SE NAO ESTIVER VAZIA, SALTA PARA A CONVERSAO
    JNE CONVERTER2

    ;SE ESTIVER VAZIA, MOSTRAR A MENSAGEM DE FRASE VAZIA
    MOV DX, OFFSET VAZIO1
    MOV AH, 9H
    INT 21H

    ;E VOLTAR A PEDIR A FRASE (VOLTA PARA O INICIO DESTA OPCAO)
    JMP OPCAO2

CONVERTER2:

    ;MOSTRAR A MENSAGEM QUE ANTECEDE O RESULTADO DA CONVERSAO
    MOV DX, OFFSET RESULTADO
    MOV AH, 9H
    INT 21H

    ;O SI VAI APONTAR PARA O PRIMEIRO CARACTER DA FRASE (POSICAO 2 DA VARIAVEL "RECEBER")
    MOV SI, OFFSET RECEBER + 2

    ;O CX VAI SER O CONTADOR DO CICLO, COM A QUANTIDADE DE CARACTERES QUE O UTILIZADOR DIGITOU (POSICAO 1)
    MOV CH, 0
    MOV CL, RECEBER+1

;CICLO QUE PERCORRE A FRASE CARACTER A CARACTER
CICLO2:

    ;COLOCAR NO DL O CARACTER ATUAL (O DL `E O REGISTRO QUE O 02H USA PARA MOSTRAR O CARACTER)
    MOV DL, [SI]

    ;COMPARAR O CARACTER COM 'A'
    CMP DL, 'A'

    ;SE FOR MENOR QUE 'A' NAO `E MAIUSCULA, ENTAO FICA IGUAL
    JB SEGUIR2

    ;COMPARAR O CARACTER COM 'Z'
    CMP DL, 'Z'

    ;SE FOR MAIOR QUE 'Z' NAO `E MAIUSCULA, ENTAO FICA IGUAL
    JA SEGUIR2

    ;SE CHEGOU AQUI, ESTA ENTRE 'A' (41H) E 'Z' (5AH), ENTAO `E MAIUSCULA
    ;A DIFERENCA ENTRE A MINUSCULA E A MAIUSCULA `E SEMPRE 20H (32), ENTAO SOMAMOS 20H
    ADD DL, 20H

    ;GUARDAR O CARACTER CONVERTIDO NO MESMO LUGAR DA STRING
    MOV [SI], DL

SEGUIR2:

    ;MOSTRAR O CARACTER (CONVERTIDO OU IGUAL), POIS AS LETRAS FORAM CONVERTIDAS E O RESTO (NUMEROS, ESPACOS, SIMBOLOS) FICOU INTACTO
    ;USAMOS O 02H E NAO O 09H, PORQUE SE A FRASE TIVER O SIMBOLO '$' O 09H PARAVA NELE
    MOV AH, 2H
    INT 21H

    ;AVANCAR PARA O PROXIMO CARACTER
    INC SI

    ;O LOOP SUBTRAI 1 AO CX E, ENQUANTO O CX NAO CHEGAR A 0, VOLTA PARA O CICLO
    LOOP CICLO2

    ;MOSTRAR A MENSAGEM DA QUANTIDADE DE CARACTERES PROCESSADOS
    MOV DX, OFFSET CONTAGEM
    MOV AH, 9H
    INT 21H

    ;A QUANTIDADE VAI DE 1 A 80 E O 02H SO MOSTRA UM CARACTER DE CADA VEZ, ENTAO SEPARAMOS EM DEZENAS E UNIDADES
    ;DIVIDIMOS POR 10: O QUOCIENTE (AL) SAO AS DEZENAS E O RESTO (AH) SAO AS UNIDADES
    MOV AL, RECEBER+1
    MOV AH, 0
    MOV BL, 10
    DIV BL

    ;GUARDAR AS UNIDADES NO BH, POIS O AH VAI SER USADO PARA CHAMAR O 02H
    MOV BH, AH

    ;SE AS DEZENAS FOREM 0, NAO MOSTRAR O ZERO A ESQUERDA (EX: 5 EM VEZ DE 05)
    CMP AL, 0
    JE UNIDADES2

    ;MOSTRAR AS DEZENAS, SOMANDO 48 ('0') PARA TRANSFORMAR O NUMERO NO CARACTER ASCII
    MOV DL, AL
    ADD DL, '0'
    MOV AH, 2H
    INT 21H

UNIDADES2:

    ;MOSTRAR AS UNIDADES, TAMBEM SOMANDO 48 ('0')
    MOV DL, BH
    ADD DL, '0'
    MOV AH, 2H
    INT 21H

    JMP QUESTAO

;O TAL AUXILIAR
MEIO:
    JMP INICIO

OPCAO1:

    ;MOSTRAR AQUELA SETA DO MENU
    MOV DX, OFFSET MENU6
    MOV AH, 9H
    INT 21H


;AQUI FICARA A LOGICA DA CONVERSAO

    ;MOSTRAR A MENSAGEM QUE PEDE A FRASE AO UTILIZADOR (O 0AH ACEITA NO MAXIMO 80 CARACTERES)
    MOV DX, OFFSET PEDIR
    MOV AH, 9H
    INT 21H

    ;RECEBER A STRING
    ;O 0AH PRECISA QUE O DX APONTE PARA O INICIO DA VARIAVEL "RECEBER", SE NAO ELE GUARDA A STRING NO SITIO ERRADO
    MOV DX, OFFSET RECEBER
    MOV AH, 0AH
    INT 21H

    ;SALTAR PARA A LINHA DE BAIXO, PARA A ENTRADA E A SAIDA FICAREM EM LINHAS DIFERENTES
    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H

    ;O 0AH GUARDA NA POSICAO 1 DA VARIAVEL "RECEBER" A QUANTIDADE DE CARACTERES QUE O UTILIZADOR DIGITOU
    ;(A POSICAO 0 TEM O MAXIMO E A STRING COMECA NA POSICAO 2)
    ;COMPARAMOS ESSE VALOR COM 0 PARA SABER SE A FRASE ESTA VAZIA (SO DIGITOU ENTER)
    MOV AL, RECEBER+1
    CMP AL, 0

    ;SE NAO ESTIVER VAZIA, SALTA PARA A CONVERSAO
    JNE CONVERTER1

    ;SE ESTIVER VAZIA, MOSTRAR A MENSAGEM DE FRASE VAZIA
    MOV DX, OFFSET VAZIO1
    MOV AH, 9H
    INT 21H

    ;E VOLTAR A PEDIR A FRASE (VOLTA PARA O INICIO DESTA OPCAO)
    JMP OPCAO1

CONVERTER1:

    ;MOSTRAR A MENSAGEM QUE ANTECEDE O RESULTADO DA CONVERSAO
    MOV DX, OFFSET RESULTADO
    MOV AH, 9H
    INT 21H

    ;O SI VAI APONTAR PARA O PRIMEIRO CARACTER DA FRASE (POSICAO 2 DA VARIAVEL "RECEBER")
    MOV SI, OFFSET RECEBER + 2

    ;O CX VAI SER O CONTADOR DO CICLO, COM A QUANTIDADE DE CARACTERES QUE O UTILIZADOR DIGITOU (POSICAO 1)
    MOV CH, 0
    MOV CL, RECEBER+1

;CICLO QUE PERCORRE A FRASE CARACTER A CARACTER
CICLO1:

    ;COLOCAR NO DL O CARACTER ATUAL (O DL `E O REGISTRO QUE O 02H USA PARA MOSTRAR O CARACTER)
    MOV DL, [SI]

    ;COMPARAR O CARACTER COM 'a'
    CMP DL, 'a'

    ;SE FOR MENOR QUE 'a' NAO `E MINUSCULA, ENTAO FICA IGUAL
    JB SEGUIR1

    ;COMPARAR O CARACTER COM 'z'
    CMP DL, 'z'

    ;SE FOR MAIOR QUE 'z' NAO `E MINUSCULA, ENTAO FICA IGUAL
    JA SEGUIR1

    ;SE CHEGOU AQUI, ESTA ENTRE 'a' (61H) E 'z' (7AH), ENTAO `E MINUSCULA
    ;A DIFERENCA ENTRE A MINUSCULA E A MAIUSCULA `E SEMPRE 20H (32), ENTAO SUBTRAIMOS 20H
    SUB DL, 20H

    ;GUARDAR O CARACTER CONVERTIDO NO MESMO LUGAR DA STRING
    MOV [SI], DL

SEGUIR1:

    ;MOSTRAR O CARACTER (CONVERTIDO OU IGUAL), POIS AS LETRAS FORAM CONVERTIDAS E O RESTO (NUMEROS, ESPACOS, SIMBOLOS) FICOU INTACTO
    ;USAMOS O 02H E NAO O 09H, PORQUE SE A FRASE TIVER O SIMBOLO '$' O 09H PARAVA NELE
    MOV AH, 2H
    INT 21H

    ;AVANCAR PARA O PROXIMO CARACTER
    INC SI

    ;O LOOP SUBTRAI 1 AO CX E, ENQUANTO O CX NAO CHEGAR A 0, VOLTA PARA O CICLO
    LOOP CICLO1

    ;MOSTRAR A MENSAGEM DA QUANTIDADE DE CARACTERES PROCESSADOS
    MOV DX, OFFSET CONTAGEM
    MOV AH, 9H
    INT 21H

    ;A QUANTIDADE VAI DE 1 A 80 E O 02H SO MOSTRA UM CARACTER DE CADA VEZ, ENTAO SEPARAMOS EM DEZENAS E UNIDADES
    ;DIVIDIMOS POR 10: O QUOCIENTE (AL) SAO AS DEZENAS E O RESTO (AH) SAO AS UNIDADES
    MOV AL, RECEBER+1
    MOV AH, 0
    MOV BL, 10
    DIV BL

    ;GUARDAR AS UNIDADES NO BH, POIS O AH VAI SER USADO PARA CHAMAR O 02H
    MOV BH, AH

    ;SE AS DEZENAS FOREM 0, NAO MOSTRAR O ZERO A ESQUERDA (EX: 5 EM VEZ DE 05)
    CMP AL, 0
    JE UNIDADES1

    ;MOSTRAR AS DEZENAS, SOMANDO 48 ('0') PARA TRANSFORMAR O NUMERO NO CARACTER ASCII
    MOV DL, AL
    ADD DL, '0'
    MOV AH, 2H
    INT 21H

UNIDADES1:

    ;MOSTRAR AS UNIDADES, TAMBEM SOMANDO 48 ('0')
    MOV DL, BH
    ADD DL, '0'
    MOV AH, 2H
    INT 21H

    JMP QUESTAO

OPCAO3:

    MOV DX, OFFSET DADOS1
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET DADOS2
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET DADOS3
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET DADOS4
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET DADOS5
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET DADOS6
    MOV AH, 9H
    INT 21H

    JMP QUESTAO

;SERVE PARA CASO ELE DIGITE UMA OPCAO INVALIDA AO LHE QUESTIONAREM SE DESEJA CONTINUAR OU NAO
ERRO2:

    MOV DX, OFFSET ERRO0
    MOV AH, 9H
    INT 21H

    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H



    ;DEPOIS DE MOSTRAR O ERROELE IRA AUTOMATICAMENTE PARA METODO QUESTAO, POIS ESTA LOGO ABAIXO
    ;E O SISTEMA COMO FAZ LINHA A LINHA, NEM PRECISAMOS COLOCAR UM JUMP PARA O METODO QUESTAO

QUESTAO:

    ;QUESTIONA SE PRETENDE CONTINUAR OU NAO
    MOV DX, OFFSET QUESTIONAR
    MOV AH, 9H
    INT 21H

    ;RECEBER O CARACTER QUE ELE DIGITOU
    MOV AH, 1H
    INT 21H

    MOV DX, OFFSET LINHA
    MOV AH, 9H
    INT 21H

    ;COMPARAR PARA VER SE OQUE ELE DIGITOU `E 's' EM MINUSCULO
    CMP AL, 's'
    
    ;SE FOR 's' SALTA PARA O METODO MEIO, QUE ELE SERA RESPONSAVEL POR SALTAR PARA O INICIO
    ;SE  TENTARMOS LOGO UM "JMP INICIO" O DOSBOX DARA UM ERRO DIZENDO:
    ; "METODO OUT OF RANGE = METODO FORA DE ALCANCE" 
    ;POIS ELE NAO CONSEGUE SALTAR PARA TAO LONGE ASSIM, ENTAO PRECISA DE UM AUXILIAR PARA JUMPAR POR ELE
    JE MEIO

    ;SE NAO FOR, ELE IRA VIR PARA AQUI E VERIFICAR SE `E 'S' EM MAIUCO
    CMP AL, 'S'

    ;SE FOR, SALTA PARA O METODO MEIO, ONDE ELE `E RESPONSAVEL POR SALTAR PARA O INICIO
    JE MEIO

    ;SE NAO FOR SIM, ENTAO COMPARAR PARA VER SE FOI N OU ALGO INVALIDO
    CMP AL, 'N'

    ;SE FOR, SALTA PARA FIM
    JE FIM

    ;SE NAO FOR 'N', VER SE FOI 'n'
    CMP AL, 'n'

    JE FIM

    ;SE NAO FOR NEM (s/S) NEM (n/N) ENTAO CONSIDERAMOS QUE INTRODUZIO ALGO INVALIDO, SEJA ELE UM CARACTER DIFERENTE OU ALGO VAZIO
    JMP ERRO2


FIM:

    MOV DX, OFFSET DESPEDIDAS
    MOV AH, 9H
    INT 21H

    ;FUNCAO PARA DEVOLVER O CONTROLE AO SISTEMA OPERACIONAL (SE NAO O DOSBOX NAO IRA SAIR DO PROGRAMA)
    MOV AX, 4C00H
    INT 21H

MAIN ENDP
END MAIN
